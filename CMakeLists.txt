 # Only the native finally boundary uses Microsoft SEH. Preserve the existing
 # MinGW C++ build and its runtime; a whole toolchain migration is unnecessary.

  find_program(SS2VR_SEH_CLANG NAMES clang REQUIRED)
  find_program(SS2VR_SEH_OBJCOPY NAMES i686-w64-mingw32-objcopy REQUIRED)
  set(SS2VR_FINALLY_OBJECT "${CMAKE_BINARY_DIR}/native-finally.o")
  add_custom_command(OUTPUT "${SS2VR_FINALLY_OBJECT}"
   COMMAND "${SS2VR_SEH_CLANG}" --target=i686-pc-windows-msvc -fms-extensions -fexceptions -O2 -Wall -Wextra -Werror
           -c "${CMAKE_SOURCE_DIR}/src/game/native_finally.c" -o "${CMAKE_BINARY_DIR}/native-finally-raw.o"
   # GNU PE ld does not consume the COFF SafeSEH symbol-index list. Keep actual
   # scope table/code relocations; existing DLL remains a legacy SEH image.
   COMMAND "${SS2VR_SEH_OBJCOPY}" --remove-section=.sxdata
           "${CMAKE_BINARY_DIR}/native-finally-raw.o" "${SS2VR_FINALLY_OBJECT}"
   DEPENDS src/game/native_finally.c VERBATIM)
  set_source_files_properties("${SS2VR_FINALLY_OBJECT}" PROPERTIES GENERATED TRUE EXTERNAL_OBJECT TRUE)
  add_custom_target(native_finally_object DEPENDS "${SS2VR_FINALLY_OBJECT}")
